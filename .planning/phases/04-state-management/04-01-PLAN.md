**Execution State Schema (schemas/execution-state.schema.yaml):**
Based on 04-RESEARCH.md "Schema Recommendations" section.

```yaml
$schema: "http://json-schema.org/draft-07/schema#"
$id: "https://ptf.dev/schemas/execution-state.schema.yaml"
title: ExecutionState
description: Master state file for PTF execution - single source of truth for "where are we?"
type: object
required:
  - status
  - current_wave
  - waves_total
  - progress
properties:
  status:
    type: string
    enum: [pending, running, paused, completed, failed]
    description: Overall execution status
  current_wave:
    type: integer
    minimum: 1
    description: Wave currently executing or next to execute
  waves_total:
    type: integer
    minimum: 1
    description: Total number of waves in plan
  progress:
    $ref: "#/definitions/Progress"
  wave_summary:
    type: object
    additionalProperties:
      type: string
      enum: [pending, running, completed, partial, failed]
    description: Status by wave number (1, 2, 3...)
  session:
    $ref: "#/definitions/Session"
  blockers:
    type: array
    items:
      type: string
    default: []
    description: List of blocking issues preventing progress
  recent_events:
    type: array
    items:
      $ref: "#/definitions/RecentEvent"
    maxItems: 10
    description: Last 10 events for quick status display

definitions:
  Progress:
    type: object
    required:
      - tasks_total
      - tasks_completed
      - tasks_pending
    properties:
      tasks_total:
        type: integer
        minimum: 0
      tasks_completed:
        type: integer
        minimum: 0
      tasks_running:
        type: integer
        minimum: 0
        default: 0
      tasks_pending:
        type: integer
        minimum: 0
      tasks_failed:
        type: integer
        minimum: 0
        default: 0
      tasks_blocked:
        type: integer
        minimum: 0
        default: 0

  Session:
    type: object
    properties:
      id:
        type: string
        pattern: "^session-[0-9]+$"
      started:
        type: string
        format: date-time
      last_update:
        type: string
        format: date-time

  RecentEvent:
    type: object
    required:
      - timestamp
      - event
    properties:
      timestamp:
        type: string
        format: date-time
      event:
        type: string
      task:
        type: string
      wave:
        type: integer
```

**Task State Schema (schemas/task-state.schema.yaml):**

```yaml
$schema: "http://json-schema.org/draft-07/schema#"
$id: "https://ptf.dev/schemas/task-state.schema.yaml"
title: TaskState
description: Per-task execution state stored in .orchestrator/state/tasks/{task-id}.yaml
type: object
required:
  - task_id
  - wave
  - status
properties:
  task_id:
    type: string
    pattern: "^[a-z0-9-]+$"
  wave:
    type: integer
    minimum: 1
  status:
    type: string
    enum: [pending, ready, running, completed, failed, blocked, skipped]
    default: pending
  attempts:
    type: array
    items:
      $ref: "#/definitions/Attempt"
    default: []
  outputs_produced:
    type: array
    items:
      $ref: "#/definitions/ProducedOutput"
    default: []
  verification:
    $ref: "#/definitions/Verification"
  blocked_by:
    type: array
    items:
      type: string
    description: Task IDs this task is waiting for
  error:
    type: string
    description: Error message if status is failed

definitions:
  Attempt:
    type: object
    required:
      - attempt
      - started
      - status
    properties:
      attempt:
        type: integer
        minimum: 1
      started:
        type: string
        format: date-time
      completed:
        type: string
        format: date-time
      status:
        type: string
        enum: [running, completed, failed]
      duration_seconds:
        type: number
      error:
        type: string

  ProducedOutput:
    type: object
    required:
      - path
    properties:
      path:
        type: string
      checksum:
        type: string
        pattern: "^sha256:[a-f0-9]{64}$"
      size_bytes:
        type: integer

  Verification:
    type: object
    properties:
      status:
        type: string
        enum: [pending, passed, failed, skipped]
        default: pending
      results:
        type: array
        items:
          type: object
          properties:
            type:
              type: string
            target:
              type: string
            passed:
              type: boolean
            message:
              type: string
```

**Wave State Schema (schemas/wave-state.schema.yaml):**

```yaml
$schema: "http://json-schema.org/draft-07/schema#"
$id: "https://ptf.dev/schemas/wave-state.schema.yaml"
title: WaveState
description: Per-wave execution state stored in .orchestrator/state/waves/wave-{N}.yaml
type: object
required:
  - wave
  - status
  - tasks
properties:
  wave:
    type: integer
    minimum: 1
    description: Wave number
  status:
    type: string
    enum: [pending, running, completed, partial, failed]
    default: pending
    description: Wave completion status (partial = some tasks failed)
  tasks:
    type: array
    minItems: 1
    items:
      $ref: "#/definitions/WaveTask"
  depends_on_waves:
    type: array
    items:
      type: integer
    default: []
    description: Wave numbers that must complete before this wave
  started_at:
    type: string
    format: date-time
  completed_at:
    type: string
    format: date-time
  duration_seconds:
    type: number
    minimum: 0
  checkpoint_file:
    type: string
    description: Path to checkpoint file if created

definitions:
  WaveTask:
    type: object
    required:
      - task_id
      - status
    properties:
      task_id:
        type: string
      status:
        type: string
        enum: [pending, running, completed, failed, skipped]
      duration_seconds:
        type: number
```

**Artifact Manifest Schema (schemas/artifact-manifest.schema.yaml):**
Based on 04-RESEARCH.md "Pattern 4: Artifact Manifest for Resume Validation".

```yaml
$schema: "http://json-schema.org/draft-07/schema#"
$id: "https://ptf.dev/schemas/artifact-manifest.schema.yaml"
title: ArtifactManifest
description: Registry of all produced artifacts for resume validation
type: object
required:
  - artifacts
properties:
  artifacts:
    type: array
    items:
      $ref: "#/definitions/ManifestEntry"
    default: []
  pending:
    type: array
    items:
      $ref: "#/definitions/PendingArtifact"
    default: []
    description: Artifacts expected but not yet produced
  last_updated:
    type: string
    format: date-time

definitions:
  ManifestEntry:
    type: object
    required:
      - path
      - type
      - produced_by
      - checksum
    properties:
      path:
        type: string
        description: File path relative to project root
      type:
        type: string
        enum: [source-code, config, test, migration, documentation, data, schema]
      produced_by:
        type: string
        description: Task ID that created this artifact
      produced_at:
        type: string
        format: date-time
      verified:
        type: boolean
        default: false
      checksum:
        type: string
        pattern: "^sha256:[a-f0-9]{64}$"
        description: SHA-256 hash for integrity verification
      consumed_by:
        type: array
        items:
          type: string
        description: Task IDs that use this artifact as input

  PendingArtifact:
    type: object
    required:
      - path
      - expected_from
    properties:
      path:
        type: string
      expected_from:
        type: string
        description: Task ID expected to produce this
      needed_by:
        type: array
        items:
          type: string
        description: Task IDs waiting for this artifact
```

Ensure checksum pattern includes "sha256:" prefix per research recommendation.
